function showTab(id,btn){
  document.querySelectorAll(".editor").forEach(e=>e.classList.remove("active"));
  document.querySelectorAll(".tab").forEach(e=>e.classList.remove("active"));
  document.getElementById(id).classList.add("active");
  btn.classList.add("active");
}

function buildDocument(){
  let html=document.getElementById("htmlCode").value;
  let css=document.getElementById("cssCode").value;
  let js=document.getElementById("jsCode").value;
  if(!/<head[\s>]/i.test(html)) html="<head></head>"+html;
  if(!/<style[\s>]/i.test(html)){
    html=html.replace(/<\/head>/i,"<style>"+css+"</style></head>");
  }
  if(js.trim() && !/JavaScript is not required/i.test(js) && !/<script[\s>]/i.test(html)){
    html=html.replace(/<\/body>/i,"<script>"+js.replace(/<\/script/gi,"<\\/script")+"</script></body>");
  }
  return html;
}

function runCode(){
  const frame=document.getElementById("output");
  frame.srcdoc=buildDocument();
  frame.onload=runTests;
}

function runTests(){
  const doc=document.getElementById("output").contentDocument;
  const list=document.getElementById("testList");
  let score=0,total=0;
  list.innerHTML="";
  tests.forEach((t,i)=>{
    let ok=false;
    try{ok=!!t.check(doc)}catch(e){ok=false}
    total+=t.marks;
    if(ok)score+=t.marks;
    const row=document.createElement("div");
    row.className="test";
    row.innerHTML=
      `<div class="status ${ok?'pass':'fail'}">${ok?'✓':'✗'}</div>`+
      `<div>${i+1}. ${t.name}</div>`+
      `<div class="marks">${t.marks}</div>`;
    list.appendChild(row);
  });
  document.getElementById("score").textContent=`${score} / ${total}`;
  document.getElementById("totalScore").textContent=`${score} / ${total}`;
}

function resetCode(){
  location.reload();
}

function submitTest(){
  runCode();
  setTimeout(()=>{
    alert("Test submitted. Your current score is "+document.getElementById("totalScore").textContent+".");
  },500);
}

window.onload=runCode;
